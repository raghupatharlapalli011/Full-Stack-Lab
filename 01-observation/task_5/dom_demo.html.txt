  <div class="box">
    <h2>DOM Manipulation Demo</h2>

    <!-- Target element whose Content, Style, and Attributes will change -->
    <div id="target-element" title="Original Title">
      Original Content: Hello World!
    </div>

    <!-- Link to demonstrate attribute modification -->
    <div>
      <a id="target-link" href="https://google.com" target="_blank">Visit Search Engine (Google)</a>
    </div>
    <br>

    <!-- Action Buttons -->
    <button onclick="changeContent()">1. Change Content</button>
    <button onclick="changeStyle()">2. Change Style</button>
    <button onclick="changeAttribute()">3. Change Attribute</button>
  </div>

  <script>
    const element = document.getElementById('target-element');
    const link = document.getElementById('target-link');

    // 1. Change CONTENT using innerHTML / textContent
    function changeContent() {
      element.innerHTML = "<strong>New Content:</strong> DOM content changed dynamically!";
    }

    // 2. Change STYLE using style property
    function changeStyle() {
      element.style.color = "white";
      element.style.backgroundColor = "#28a745";
      element.style.fontSize = "22px";
      element.style.border = "2px solid #1e7e34";
    }

    // 3. Change ATTRIBUTES using setAttribute
    function changeAttribute() {
      // Changes link destination and text
      link.setAttribute('href', 'https://github.com');
      link.innerText = 'Visit GitHub (Attribute & Link Changed!)';
      element.setAttribute('title', 'This tooltip attribute was updated by JS!');
      alert('Attribute changed! Hover on the box or click the link.');
    }
  </script>
